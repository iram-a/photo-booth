  <script>
    const videoElement = document.getElementById('video-feed');
    const canvasElement = document.getElementById('photo-canvas');
    const startButton = document.getElementById('start-btn');
    const captureButton = document.getElementById('capture-btn');
    const recaptureButton = document.getElementById('recapture-btn');
    const photoSlots = [
      document.getElementById('photo-1'),
      document.getElementById('photo-2'),
      document.getElementById('photo-3'),
      document.getElementById('photo-4')
    ];

    let stream;
    let captureCount = 0;

    async function startCamera() {
      if (stream) return;
      try {
        stream = await navigator.mediaDevices.getUserMedia({ video: { width: 1280, height: 720 } });
        videoElement.srcObject = stream;
        videoElement.play();
        startButton.disabled = true;
        captureButton.disabled = false;
      } catch (err) {
        console.error("CRITICAL ERROR accessing the camera: ", err.name, err.message);
        alert("Could not access the camera. Please check your browser permissions.");
      }
    }

    function capturePhoto() {
      if (captureCount >= 4) return;
      const context = canvasElement.getContext('2d');
      canvasElement.width = videoElement.videoWidth;
      canvasElement.height = videoElement.videoHeight;
      context.drawImage(videoElement, 0, 0, canvasElement.width, canvasElement.height);
      const dataUrl = canvasElement.toDataURL('image/png');
      photoSlots[captureCount].src = dataUrl;
      captureCount++;
      recaptureButton.disabled = captureCount === 0;
      captureButton.disabled = captureCount >= 4;
    }

    function recapturePhotos() {
      if (captureCount === 0) return;
      captureCount--;
      photoSlots[captureCount].src = '';
      recaptureButton.disabled = captureCount === 0;
      captureButton.disabled = false;
    }

    startButton.addEventListener('click', startCamera);
    captureButton.addEventListener('click', capturePhoto);
    recaptureButton.addEventListener('click', recapturePhotos);
  </script>
